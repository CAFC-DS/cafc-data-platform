"""
Configuration for Impect API and Snowflake connections
"""
import os
from dotenv import load_dotenv

# Load environment variables
load_dotenv()

# Impect API Configuration
IMPECT_BASE_URL = os.getenv("IMPECT_BASE_URL", "https://api.impect.com")
IMPECT_TOKEN_URL = os.getenv("IMPECT_TOKEN_URL", "https://login.impect.com/auth/realms/production/protocol/openid-connect/token")
IMPECT_USERNAME = os.getenv("IMPECT_USERNAME")
IMPECT_PASSWORD = os.getenv("IMPECT_PASSWORD")

# Default iteration ID
DEFAULT_ITERATION_ID = 1410

# Snowflake Configuration
SNOWFLAKE_ACCOUNT = os.getenv("SNOWFLAKE_ACCOUNT")
SNOWFLAKE_USER = os.getenv("SNOWFLAKE_USER")
SNOWFLAKE_PASSWORD = os.getenv("SNOWFLAKE_PASSWORD")
SNOWFLAKE_DATABASE = os.getenv("SNOWFLAKE_DATABASE")
SNOWFLAKE_SCHEMA = os.getenv("SNOWFLAKE_SCHEMA", "PUBLIC")
SNOWFLAKE_WAREHOUSE = os.getenv("SNOWFLAKE_WAREHOUSE")
SNOWFLAKE_ROLE = os.getenv("SNOWFLAKE_ROLE")

# Request settings
REQUEST_TIMEOUT = 30
MAX_RETRIES = 3

# Token cache settings
TOKEN_CACHE_FILE = ".impect_token_cache.json"
TOKEN_EXPIRY_HOURS = 24

================================================================================
KEY FEATURES IMPLEMENTED
================================================================================

✓ OAuth2 Authentication
  - Automatic token fetching using username/password
  - Token caching for 24 hours in .impect_token_cache.json
  - Automatic token refresh when expired

✓ Rate Limiting
  - Handles 429 errors with 1-second delays
  - Exponential backoff for failed requests

✓ API Endpoints (All default to iteration 1410)
  - get_iterations() - Get all iterations
  - get_matches(iteration_id) - Get matches
  - get_squads(iteration_id) - Get squads
  - get_players(iteration_id) - Get players
  - get_coaches(iteration_id) - Get coaches
  - get_stadiums(iteration_id) - Get stadiums
  - get_countries() - Get all countries

✓ Snowflake Loading
  - load_json_to_table() - Load JSON to table
  - create_table_from_json() - Create new table with data
  - append_to_table() - Append to existing table
  - load_dataframe_to_table() - Load pandas DataFrame

✓ Simple Procedural Style
  - No complex classes or patterns
  - Easy to understand and modify
  - Clear function names

================================================================================
USAGE EXAMPLES
================================================================================

Example 1: Fetch Data from Impect
----------------------------------
from impect_api import get_matches, get_players

# Get matches for iteration 1410 (default)
matches = get_matches()

# Get players for a specific iteration
players = get_players(iteration_id=1234)


Example 2: Load to Snowflake
-----------------------------
from snowflake_loader import create_table_from_json

# Create table and load data
create_table_from_json(matches, 'impect_matches')


Example 3: Complete ETL Pipeline
---------------------------------
from impect_api import get_matches, get_players
from snowflake_loader import create_table_from_json

# Extract
print("Fetching data from Impect...")
matches = get_matches(iteration_id=1410)
players = get_players(iteration_id=1410)

# Load
print("Loading to Snowflake...")
create_table_from_json(matches, 'impect_matches')
create_table_from_json(players, 'impect_players')

print("ETL complete!")
